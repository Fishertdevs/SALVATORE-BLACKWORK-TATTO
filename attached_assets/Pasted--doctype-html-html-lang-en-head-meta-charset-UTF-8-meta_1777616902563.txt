    <script>
      // ══════════════════════════════════════════
      //  ASCII CONFIG
      // ══════════════════════════════════════════
      const ASCII_CHARS       = "........:::=+xX#0369";
      const FONT_SIZE         = 14;
      const ASPECT_W          = 4;
      const ASPECT_H          = 5;
      const ASCII_COLS        = 25;
      const IMAGE_STAGGER_MS  = 70;
      const CELL_APPEAR_MS    = 2;
      const SCRAMBLE_COUNT    = 10;
      const SCRAMBLE_SPEED_MS = 100;

      const denseCharIndex = ASCII_CHARS.lastIndexOf(".");
      const denseChars     = ASCII_CHARS.slice(denseCharIndex + 1).split("");

      const measureCtx = document.createElement("canvas").getContext("2d");
      measureCtx.font  = `${FONT_SIZE}px monospace`;
      const charWidth  = Math.ceil(measureCtx.measureText("M").width);
      const charHeight = FONT_SIZE;
      const ASCII_ROWS = Math.round(ASCII_COLS * (ASPECT_H / ASPECT_W) * (charWidth / charHeight));

      // ══════════════════════════════════════════
      //  BUILD GRID
      // ══════════════════════════════════════════
      const FEATURED_INDEX = 6; // 0-based
      const portraitSeeds  = [10,25,37,48,62,75,88,99,113,127,140,155,168,182,197];
      const featuredSeed   = 210;

      const gallery = document.getElementById("gallery-layer");
      let featuredCard = null;

      for (let i = 0; i < 16; i++) {
        const div = document.createElement("div");

        if (i === FEATURED_INDEX) {
          div.className = "img featured";
          const img = document.createElement("img");
          img.id          = "featured-img-el";
          img.crossOrigin = "anonymous";
          img.src         = `https://picsum.photos/seed/${featuredSeed}/1600/900`;
          div.appendChild(img);
          featuredCard = div;
        } else {
          const seedIdx = i < FEATURED_INDEX ? i : i - 1;
          div.className = "img";
          const img = document.createElement("img");
          img.className   = "ascii-reveal";
          img.crossOrigin = "anonymous";
          img.src         = `https://picsum.photos/seed/${portraitSeeds[seedIdx]}/400/500`;
          div.appendChild(img);
        }

        gallery.appendChild(div);
      }

      // ══════════════════════════════════════════
      //  ASCII EFFECTS
      // ══════════════════════════════════════════
      document.querySelectorAll("img.ascii-reveal").forEach((img, i) => {
        const canvas = document.createElement("canvas");
        img.closest(".img").appendChild(canvas);
        const delay = i * IMAGE_STAGGER_MS;
        const run   = () => runEffect(img, canvas, delay, ASCII_COLS, ASCII_ROWS, ASPECT_W, ASPECT_H);
        img.complete && img.naturalWidth ? run() : img.addEventListener("load", run);
      });

      const featuredImgEl = document.getElementById("featured-img-el");
      (() => {
        const canvas = document.createElement("canvas");
        featuredCard.appendChild(canvas);
        const run = () => runEffectLeftCrop(featuredImgEl, canvas, 500, () => {
          featuredCard.classList.add("revealed");
        });
        featuredImgEl.complete && featuredImgEl.naturalWidth ? run() : featuredImgEl.addEventListener("load", run);
      })();

      function runEffectLeftCrop(img, canvas, delay, onDone) {
        const targetAspect = ASPECT_W / ASPECT_H; 
        const cropW = img.naturalHeight * targetAspect; 
        const cropX = 0; 
        const cropY = 0;
        const cropH = img.naturalHeight;

        const sc = document.createElement("canvas");
        sc.width = ASCII_COLS; sc.height = ASCII_ROWS;
        sc.getContext("2d").drawImage(img, cropX, cropY, cropW, cropH, 0, 0, ASCII_COLS, ASCII_ROWS);
        const { data } = sc.getContext("2d").getImageData(0, 0, ASCII_COLS, ASCII_ROWS);

        const ag = [], bg = [];
        for (let r = 0; r < ASCII_ROWS; r++) {
          const ar = [], br = [];
          for (let c = 0; c < ASCII_COLS; c++) {
            const pi = (r * ASCII_COLS + c) * 4;
            const lum = (data[pi]*.299 + data[pi+1]*.587 + data[pi+2]*.114) / 255;
            const ci = Math.min(ASCII_CHARS.length-1, Math.floor((1-lum)*ASCII_CHARS.length));
            ar.push(ASCII_CHARS[ci]); br.push(ci);
          }
          ag.push(ar); bg.push(br);
        }
        prepareCanvas(canvas, ASCII_COLS, ASCII_ROWS);
        animateCells(canvas, ag, bg, delay, ASCII_COLS, ASCII_ROWS, onDone);
      }

      function runEffect(img, canvas, delay, cols, rows, aw, ah, onDone) {
        const { asciiGrid, brightnessGrid } = imageToAsciiGrid(img, cols, rows, aw, ah);
        prepareCanvas(canvas, cols, rows);
        animateCells(canvas, asciiGrid, brightnessGrid, delay, cols, rows, onDone);
      }

      function imageToAsciiGrid(img, cols, rows, aw, ah) {
        const ia = img.naturalWidth / img.naturalHeight;
        const ta = aw / ah;
        let cx=0,cy=0,cw=img.naturalWidth,ch=img.naturalHeight;
        if (ia > ta) { cw=img.naturalHeight*ta; cx=(img.naturalWidth-cw)/2; }
        else         { ch=img.naturalWidth/ta;  cy=(img.naturalHeight-ch)/2; }
        const sc = document.createElement("canvas");
        sc.width=cols; sc.height=rows;
        sc.getContext("2d").drawImage(img,cx,cy,cw,ch,0,0,cols,rows);
        const {data} = sc.getContext("2d").getImageData(0,0,cols,rows);
        const ag=[],bg=[];
        for(let r=0;r<rows;r++){
          const ar=[],br=[];
          for(let c=0;c<cols;c++){
            const pi=(r*cols+c)*4;
            const lum=(data[pi]*.299+data[pi+1]*.587+data[pi+2]*.114)/255;
            const ci=Math.min(ASCII_CHARS.length-1,Math.floor((1-lum)*ASCII_CHARS.length));
            ar.push(ASCII_CHARS[ci]); br.push(ci);
          }
          ag.push(ar); bg.push(br);
        }
        return {asciiGrid:ag,brightnessGrid:bg};
      }

      function prepareCanvas(canvas,cols,rows){
        const dpr=2;
        canvas.width=cols*charWidth*dpr;
        canvas.height=rows*charHeight*dpr;
        const ctx=canvas.getContext("2d");
        ctx.fillStyle="#111";
        ctx.fillRect(0,0,canvas.width,canvas.height);
      }

      function drawChar(ctx,col,row,char){
        ctx.fillStyle="#111";
        ctx.fillRect(col*charWidth,row*charHeight,charWidth,charHeight);
        ctx.fillStyle="#c8c8c8";
        ctx.fillText(char,col*charWidth,row*charHeight);
      }

      function shuffle(arr){
        for(let i=arr.length-1;i>0;i--){
          const j=Math.floor(Math.random()*(i+1));
          [arr[i],arr[j]]=[arr[j],arr[i]];
        }
        return arr;
      }

      function animateCells(canvas,ag,bg,delay,cols,rows,onDone){
        const dpr=2;
        const ctx=canvas.getContext("2d");
        ctx.setTransform(dpr,0,0,dpr,0,0);
        ctx.font=`${charHeight}px monospace`;
        ctx.textBaseline="top";
        const total=cols*rows;
        const state=new Array(total).fill(null);
        let settled=0;
        const order=shuffle(Array.from({length:total},(_,i)=>i));

        const done=()=>{
          canvas.closest(".img")?.classList.add("revealed");
          onDone?.();
        };

        order.forEach((ci,i)=>{
          setTimeout(()=>{
            const r=Math.floor(ci/cols),c=ci%cols;
            const dark=bg[r][c]>denseCharIndex;
            if(!dark){
              drawChar(ctx,c,r,ag[r][c]);
              state[ci]=0; settled++;
              if(settled===total) done();
            } else {
              drawChar(ctx,c,r,denseChars[Math.floor(Math.random()*denseChars.length)]);
              state[ci]=SCRAMBLE_COUNT;
            }
          },delay+i*CELL_APPEAR_MS);
        });

        const tick=setInterval(()=>{
          let still=false;
          for(let ci=0;ci<total;ci++){
            const rem=state[ci];
            if(rem===null||rem===0) continue;
            still=true;
            const r=Math.floor(ci/cols),c=ci%cols;
            if(rem===1){
              drawChar(ctx,c,r,ag[r][c]);
              state[ci]=0; settled++;
              if(settled===total) done();
            } else {
              drawChar(ctx,c,r,denseChars[Math.floor(Math.random()*denseChars.length)]);
              state[ci]=rem-1;
            }
          }
          if(!still&&settled===total) clearInterval(tick);
        },SCRAMBLE_SPEED_MS);
      }

      // ══════════════════════════════════════════
      //  FIXED GSAP LOGIC — DIRECT DIMENSION ANIMATION
      // ══════════════════════════════════════════
      const fullSection = document.getElementById("fullscreen-section");
      const fullContent = document.getElementById("fullscreen-content");
      const backBtn     = document.getElementById("back-btn");
      const galleryEl   = document.getElementById("gallery-layer");

      let isExpanded = false;

      featuredCard.addEventListener("click", expandSection);

      function expandSection() {
        if (isExpanded) return;
        isExpanded = true;

        const rect = featuredCard.getBoundingClientRect();

        // Thay vì dùng scale, ta set cứng width/height bằng thẻ ảnh lúc ban đầu
        gsap.set(fullSection, {
          opacity: 1,
          pointerEvents: "all",
          width: rect.width,
          height: rect.height,
          left: rect.left,
          top: rect.top,
          borderRadius: "4px",
          x: 0, y: 0, scaleX: 1, scaleY: 1 // Clear default scale
        });
        
        // Đặt vị trí ảnh cắt chính xác khung trái của ảnh gốc 16:9
        gsap.set("#fullscreen-img", { objectPosition: "left center" });
        gsap.set(fullContent, { opacity: 0, y: 24 });
        gsap.set(backBtn, { opacity: 0, pointerEvents: "none" });

        gsap.timeline({ defaults: { ease: "expo.inOut" } })
          .to(galleryEl, { opacity: 0.12, duration: 0.5 }, 0)
          .to(featuredCard, { opacity: 0, duration: 0.2 }, 0)
          // Phóng to trực tiếp bằng Layout Dimension
          .to(fullSection, {
            width: "100vw",
            height: "100vh",
            left: 0,
            top: 0,
            borderRadius: "0px",
            duration: 0.8,
          }, 0)
          // Đồng thời dịch chuyển ảnh về giữa để hiển thị phần còn lại của ảnh 16:9
          .to("#fullscreen-img", { objectPosition: "center center", duration: 0.8, ease: "expo.inOut" }, 0)
          .to(fullContent, { opacity: 1, y: 0, duration: 0.45, ease: "power2.out" }, "-=0.2")
          .to(backBtn, { opacity: 1, duration: 0.3, ease: "power2.out", onStart: () => { backBtn.style.pointerEvents = "all"; } }, "-=0.15");
      }

      function collapseSection() {
        if (!isExpanded) return;

        const rect = featuredCard.getBoundingClientRect();

        gsap.timeline({
          defaults: { ease: "expo.inOut" },
          onComplete: () => {
            isExpanded = false;
            gsap.set(fullSection, {
              opacity: 0, pointerEvents: "none",
              width: "100%", height: "100%", top: 0, left: 0, borderRadius: "0px"
            });
          }
        })
        .to(backBtn, { opacity: 0, duration: 0.18, ease: "power2.in", onStart: () => { backBtn.style.pointerEvents = "none"; } })
        .to(fullContent, { opacity: 0, y: 16, duration: 0.22, ease: "power2.in" }, "<")
        // Thu nhỏ lại y hệt kích thước thẻ ảnh
        .to(fullSection, {
          width: rect.width,
          height: rect.height,
          left: rect.left,
          top: rect.top,
          borderRadius: "4px",
          duration: 0.72,
        }, "-=0.05")
        // Kéo khung nhìn của ảnh về lại phía bên trái
        .to("#fullscreen-img", { objectPosition: "left center", duration: 0.72 }, "<")
        .to(fullSection, { opacity: 0, duration: 0.2 }, "-=0.15")
        .to(galleryEl, { opacity: 1, duration: 0.35 }, "-=0.2")
        .to(featuredCard, { opacity: 1, duration: 0.25 }, "-=0.2");
      }

      backBtn.addEventListener("click", collapseSection);
      document.addEventListener("keydown", e => { if (e.key === "Escape") collapseSection(); });
    </script>
